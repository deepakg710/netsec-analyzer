! Architecture: Multi-Tier Campus Enterprise

!

 no ip redirects
 no ip unreachables
 no shutdown

ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! --- OSPF Dynamic Routing for Internal Subnets ---
router ospf 1
 router-id 1.1.1.1
 passive-interface GigabitEthernet0/0
 passive-interface GigabitEthernet0/1
 network 10.0.0.0 0.0.0.3 area 0
 network 10.0.0.4 0.0.0.3 area 0
 network 192.168.100.0 0.0.0.255 area 0

!
! --- Perimeter Inbound Access List ---
ip access-list extended WAN-INBOUND
 remark === Block spoofed internal IP addresses from WAN ===
 deny ip 10.0.0.0 0.255.255.255 any log
 deny ip 172.16.0.0 0.15.255.255 any log
 deny ip 192.168.0.0 0.0.255.255 any log
 deny ip 127.0.0.0 0.255.255.255 any log
 remark === Allow Inbound Web & Mail to DMZ Servers only ===
 permit tcp any host 192.168.100.10 eq 80
 permit tcp any host 192.168.100.10 eq 443
 permit tcp any host 192.168.100.25 eq 25
 remark === Allow Return Traffic (Established Sessions) ===
 permit tcp any any established
 permit icmp any any echo-reply
 remark === Explicit Deny All from Internet ===
 deny ip any any log
!
! --- DMZ Security Policy ---
ip access-list extended DMZ-FILTER
 remark === DMZ servers can only reply to WAN or query Internal DNS ===
 permit udp host 192.168.100.10 host 192.168.30.10 eq 53
 permit tcp host 192.168.100.10 host 192.168.30.20 eq 3306
 deny ip 192.168.100.0 0.0.0.255 192.168.0.0 0.0.255.255 log
 permit ip 192.168.100.0 0.0.0.255 any
!
! --- Secure Management Plane ---
line con 0
 exec-timeout 5 0
 transport output none
 login local
!
line vty 0 4
 transport input ssh
 exec-timeout 10 0
 login local
!
end
